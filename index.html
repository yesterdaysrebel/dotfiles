# #!/bin/zsh # # dotfiles installer | https://github.com/yesterdaysrebel/dotfiles # Served at https://yesterdaysrebel.github.io/dotfiles/ # Adapted from github.com/venkatamutyala/dotfiles (MIT). # # This file is a polyglot -- it works two ways: # * Install: curl -fsSL https://yesterdaysrebel.github.io/dotfiles/ | zsh # * Read it: open the URL in a browser. "#<plaintext>" on line 1 makes the # browser render the rest as raw text; zsh treats it as a comment. # # Debian/Ubuntu only (VMs, codespaces, devcontainers). Needs zsh, git and curl; # jq, vim and the `claude`/`code` CLIs are used when present. Never installs # system packages. Safe to re-run: unchanged files are left alone and anything # it would overwrite is backed up first. set -euo pipefail export GIT_TERMINAL_PROMPT=0 # never hang on a credential prompt # Where config files come from. Interactive runs prompt for a tag (via /dev/tty, so # `curl ... | zsh` works) unless DOTFILES_REF is set; with no TTY: DOTFILES_REF, then # the newest tag, then main. The ref is resolved to one commit SHA and every file is # fetched from that commit. REPO_RAW (e.g. file:///repo) skips resolution -- for tests. REPO_SLUG="yesterdaysrebel/dotfiles" REPO_GIT="${REPO_GIT:-https://github.com/${REPO_SLUG}}" REPO_RAW="${REPO_RAW:-}" DOTFILES_REF="${DOTFILES_REF:-}" # Shared GlueOps AI rules (private; needs GitHub access). Tracks main, like `ais update`. AI_RULES_REPO="${AI_RULES_REPO:-https://github.com/internal-GlueOps/ai-rules-central.git}" AI_RULES_DIR="${HOME}/.local/share/ai-rules-central" VSCODE_EXTENSION="anthropic.claude-code" # --- Pinned dependency versions (from venkatamutyala/dotfiles v0.7.0) -------- # OMZ/vim-plug use commit SHAs. zsh plugins are cloned by tag but the resolved # commit is verified against a pinned SHA (tags are mutable). OMZ_REF="3604dc23e0d95b5ce9a3932838a7b103ef5ff0c1" # ohmyzsh/ohmyzsh VIM_PLUG_REF="34467fc07d1bf1b3a6588e9d62711b9f7a8afda3" # junegunn/vim-plug ZSH_SYNTAX_HIGHLIGHTING_REF="0.8.0"; ZSH_SYNTAX_HIGHLIGHTING_SHA="db085e4661f6aafd24e5acb5b2e17e4dd5dddf3e" ZSH_AUTOSUGGESTIONS_REF="v0.7.1"; ZSH_AUTOSUGGESTIONS_SHA="e52ee8ca55bcc56a17c828767a3f98f22a68d4eb" YOU_SHOULD_USE_REF="1.11.1"; YOU_SHOULD_USE_SHA="ff371d6a11b653e1fa8dda4e61c896c78de26bfa" TS="$(date +%Y%m%d-%H%M%S)" BACKUP_DIR="${HOME}/.dotfiles-backup-${TS}" typeset -gi BACKUPS=0 UPDATED=0 typeset -ga SUMMARY RESOLVED="" CURRENT="startup" log() { print -r -- "==> $*"; } warn() { print -r -- "WARN: $*" >&2; } step() { SUMMARY+=("$(printf ' %-5s %-20s %s' "$1" "$CURRENT" "${2:-}")"); } hint() { SUMMARY+=(" -> $*"); } die() { step FAIL "$*"; print -r -- "ERROR: $*" >&2; exit 1; } print_summary() { (( ${#SUMMARY} )) || return 0 print -r -- "" print -r -- "dotfiles (${RESOLVED:-unresolved})" print -rl -- "${SUMMARY[@]}" (( BACKUPS )) && print -r -- "Backups: ${BACKUP_DIR} (${BACKUPS})" return 0 } # `exit` (from die) runs the EXIT trap, but zsh skips it when errexit fires inside a # function -- so unexpected failures print the summary from ZERR instead. trap print_summary EXIT TRAPZERR() { step FAIL "unexpected error (see above)"; print_summary; trap - EXIT; } run_cde_init() { CURRENT="cde-init" if ! command -v cde-init >/dev/null 2>&1; then step SKIP "not a GlueOps codespace"; return 0 fi log "Running cde-init (GlueOps CDE bootstrap)" # stdin is the rest of this script under `curl | zsh`; don't let cde-init read it. if cde-init </dev/null; then step OK; else step WARN "exited non-zero; continuing"; fi } os_guard() { CURRENT="os check" [[ "$(uname -s)" == "Linux" ]] || die "Linux (Debian/Ubuntu) only. Detected $(uname -s)." if [[ -r /etc/os-release ]]; then local id idlike id="$(. /etc/os-release 2>/dev/null; print -r -- "${ID:-}")" idlike="$(. /etc/os-release 2>/dev/null; print -r -- "${ID_LIKE:-}")" [[ "$id" == debian || "$id" == ubuntu || "$idlike" == *debian* ]] \ || warn "Non Debian/Ubuntu Linux (ID='$id'); continuing but untested." fi command -v git >/dev/null 2>&1 || die "git is required. Install it and re-run." command -v curl >/dev/null 2>&1 || die "curl is required. Install it and re-run." } # This repo's tags, newest first. Fails (non-zero) when offline. list_tags() { git ls-remote --tags --refs "$REPO_GIT" | awk -F/ '{print $NF}' | sort -rV } # Interactive picker. Menu goes to /dev/tty; only the chosen ref is printed to stdout. pick_tag() { local tags; tags="$(list_tags 2>/dev/null)" || tags="" if [[ -z "$tags" ]]; then warn "No tags found for ${REPO_SLUG} (none cut yet, or offline). Using 'main'." print -r -- "main"; return 0 fi local -a arr=("${(@f)tags}") arr+=("main") { print -r -- "" print -r -- "Which version of ${REPO_SLUG} do you want to install?" local i label for (( i = 1; i <= ${#arr[@]}; i++ )); do label="${arr[$i]}"; [[ "$label" == main ]] && label="main (latest, unpinned)" print -r -- " ${i}) ${label}" done print -rn -- "Enter a number [default 1 = ${arr[1]}]: " } >/dev/tty local choice="" read -r choice </dev/tty || true if [[ -z "$choice" ]]; then print -r -- "${arr[1]}" elif [[ "$choice" == <-> ]] && (( choice >= 1 && choice <= ${#arr[@]} )); then print -r -- "${arr[$choice]}" else print -r -- "$choice" # a literal tag/branch name; validated below fi } select_repo_ref() { CURRENT="version" if [[ -n "$REPO_RAW" ]]; then RESOLVED="$REPO_RAW"; log "Using REPO_RAW override: ${REPO_RAW}"; return 0 fi local ref="$DOTFILES_REF" # A real controlling terminal? Opening /dev/tty is the only reliable test. if [[ -z "$ref" ]] && { true </dev/tty; } 2>/dev/null; then ref="$(pick_tag)" elif [[ -z "$ref" ]]; then local -a tags=("${(@f)$(list_tags 2>/dev/null || true)}") ref="${tags[1]:-main}" fi # Exact refs only (ls-remote patterns tail-match, so `main` would match feat/main). local out sha out="$(git ls-remote "$REPO_GIT" "refs/tags/${ref}" "refs/tags/${ref}^{}" "refs/heads/${ref}")" \ || die "Can't reach ${REPO_GIT} to resolve '${ref}' (offline, or no access)." sha="$(print -r -- "$out" | awk '$2 ~ /\^\{\}$/ {print $1; exit}')" # peeled annotated tag [[ -n "$sha" ]] || sha="$(print -r -- "$out" | awk 'NF {print $1; exit}')" [[ -n "$sha" ]] || die "No tag or branch named '${ref}' in ${REPO_SLUG}." RESOLVED="${ref} @ ${sha[1,7]}" REPO_RAW="https://raw.githubusercontent.com/${REPO_SLUG}/${sha}" log "Installing config files from ${ref} (commit ${sha})" } backup_one() { # copy (not move) an existing target into the backup dir local t="$1"; [[ -e "$t" || -L "$t" ]] || return 0 local dest="${BACKUP_DIR}/${t#$HOME/}" mkdir -p "${dest:h}"; cp -a "$t" "$dest"; BACKUPS+=1 log " backed up: $t" } # Download to a temp file; replace the target only if the content changed. install_file() { local url="$1" dest="$2" tmp mkdir -p "${dest:h}" tmp="$(mktemp "${dest:h}/.${dest:t}.XXXXXX")" if ! curl -fsSL "$url" -o "$tmp"; then rm -f "$tmp"; die "download failed: $url"; fi if [[ -f "$dest" && ! -L "$dest" ]] && cmp -s "$tmp" "$dest"; then rm -f "$tmp"; return 0; fi backup_one "$dest" rm -f "$dest" # drop a symlink rather than writing through it mv "$tmp" "$dest"; UPDATED+=1 log " updated: $dest" } install_omz() { # pinned clone; never clobbers .zshrc or the login shell CURRENT="Oh-My-Zsh" if [[ -d ~/.oh-my-zsh ]]; then step OK "already present" else log "Installing Oh-My-Zsh (pinned @ ${OMZ_REF})" git -c advice.detachedHead=false clone --quiet https://github.com/ohmyzsh/ohmyzsh ~/.oh-my-zsh </dev/null \ || die "clone failed" git -C ~/.oh-my-zsh -c advice.detachedHead=false checkout --quiet "$OMZ_REF" step OK "installed @ ${OMZ_REF[1,7]}" fi [[ -f ~/.zshrc ]] || cp ~/.oh-my-zsh/templates/zshrc.zsh-template ~/.zshrc } fetch_configs() { CURRENT="config files" log "Fetching config files" install_file "${REPO_RAW}/tmux.conf" ~/.tmux.conf install_file "${REPO_RAW}/tmux-cheatsheet.txt" ~/.tmux-cheatsheet.txt install_file "${REPO_RAW}/vimrc" ~/.vimrc install_file "https://raw.githubusercontent.com/junegunn/vim-plug/${VIM_PLUG_REF}/plug.vim" ~/.vim/autoload/plug.vim install_file "${REPO_RAW}/dotfiles.zsh" ~/.oh-my-zsh/custom/dotfiles.zsh install_file "${REPO_RAW}/claude/CLAUDE.md" ~/.claude/CLAUDE.md step OK "${UPDATED} updated" } # Clone/pull the shared AI rules and expose each Copilot prompt as a Claude slash # command (/ai-review, ...). ~/.claude/CLAUDE.md imports the instruction files. install_ai_rules() { CURRENT="AI rules" log "Syncing GlueOps AI rules into ${AI_RULES_DIR}" local stale="" if [[ -d "$AI_RULES_DIR/.git" ]]; then git -C "$AI_RULES_DIR" pull --quiet --ff-only </dev/null || stale=" (pull failed; using existing copy)" else if [[ -e "$AI_RULES_DIR" ]]; then backup_one "$AI_RULES_DIR"; rm -rf "$AI_RULES_DIR"; fi mkdir -p "${AI_RULES_DIR:h}" if ! git clone --quiet "$AI_RULES_REPO" "$AI_RULES_DIR" </dev/null; then step WARN "can't clone ${AI_RULES_REPO:t:r} (no access?)" hint "gh auth login with an account that has access, then re-run" return 0 fi fi local cmds=~/.claude/commands f name link local -i n=0 mkdir -p "$cmds" for f in "$AI_RULES_DIR"/.github/prompts/*.prompt.md(N); do name="${${f:t}%.prompt.md}"; link="$cmds/${name}.md" if [[ -e "$link" || -L "$link" ]] && [[ "$(readlink "$link")" != "$f" ]]; then backup_one "$link"; rm -f "$link" fi ln -sfn "$f" "$link"; n+=1 done # Drop links to prompts that were removed upstream. for link in "$cmds"/*.md(N@); do [[ "$(readlink "$link")" == "$AI_RULES_DIR"/* && ! -e "$link" ]] && rm -f "$link" done local sha; sha="$(git -C "$AI_RULES_DIR" rev-parse --short HEAD)" if [[ -n "$stale" ]]; then step WARN "${n} commands @ ${sha}${stale}"; else step OK "${n} commands @ ${sha}"; fi } clone_plugin() { # re-verifies on every run, so pin bumps apply local url="$1" ref="$2" sha="$3" dir="$4" if [[ -d "$dir/.git" && "$(git -C "$dir" rev-parse HEAD 2>/dev/null)" == "$sha" ]]; then return 0 fi if [[ -e "$dir" ]]; then backup_one "$dir"; rm -rf "$dir"; fi git -c advice.detachedHead=false clone --quiet --depth 1 --branch "$ref" "$url" "$dir" </dev/null \ || die "clone failed: ${dir:t}@${ref}" local got; got="$(git -C "$dir" rev-parse HEAD)" if [[ "$got" != "$sha" ]]; then rm -rf "$dir" die "${dir:t}: tag '$ref' resolved to $got, expected $sha (moved/tampered tag?)." fi log " cloned: ${dir:t}@${ref} (verified ${sha})" } install_zsh_plugins() { CURRENT="zsh plugins" log "Installing zsh plugins" local p=~/.oh-my-zsh/custom/plugins; mkdir -p "$p" clone_plugin https://github.com/zsh-users/zsh-syntax-highlighting "$ZSH_SYNTAX_HIGHLIGHTING_REF" "$ZSH_SYNTAX_HIGHLIGHTING_SHA" "$p/zsh-syntax-highlighting" clone_plugin https://github.com/zsh-users/zsh-autosuggestions "$ZSH_AUTOSUGGESTIONS_REF" "$ZSH_AUTOSUGGESTIONS_SHA" "$p/zsh-autosuggestions" clone_plugin https://github.com/MichaelAquilina/zsh-you-should-use.git "$YOU_SHOULD_USE_REF" "$YOU_SHOULD_USE_SHA" "$p/you-should-use" if grep -q 'oh-my-zsh.sh' ~/.zshrc; then step OK "3 verified, enabled via custom/dotfiles.zsh" else step WARN "3 verified, but ~/.zshrc doesn't load Oh-My-Zsh, so they're inactive" fi } install_vim_plugins() { CURRENT="vim plugins" if ! command -v vim >/dev/null 2>&1; then step SKIP "vim not installed; later: vim +PlugInstall +qall"; return 0 fi log "Installing vim plugins (vim +PlugInstall +qall)" if vim -E -s -u ~/.vimrc +PlugInstall +qall </dev/null >/dev/null 2>&1; then step OK else step WARN "PlugInstall non-zero; re-run: vim +PlugInstall +qall" fi } # Claude Code reads CLAUDE_CODE_OAUTH_TOKEN from the environment itself (the GlueOps # Slack bot puts it in the codespace env), so there's no login to run. The token alone # doesn't skip first-run onboarding, so mark that done. Never print the token. setup_claude_auth() { CURRENT="Claude auth" if [[ -z "${CLAUDE_CODE_OAUTH_TOKEN:-}" ]]; then step WARN "CLAUDE_CODE_OAUTH_TOKEN not set" hint "add it to your profile via the Slack bot, then re-run" return 0 fi local v for v in ANTHROPIC_AUTH_TOKEN ANTHROPIC_API_KEY; do [[ -z "${(P)v:-}" ]] || warn "$v is set and takes precedence over CLAUDE_CODE_OAUTH_TOKEN." done if ! command -v jq >/dev/null 2>&1; then step WARN "token set, but jq missing (needed to skip onboarding)"; return 0 fi if ! jq -e '.hasCompletedOnboarding == true' ~/.claude.json >/dev/null 2>&1; then local tmp; tmp="$(mktemp ~/.claude.json.XXXXXX)" if [[ -f ~/.claude.json ]]; then backup_one ~/.claude.json jq '. + {"hasCompletedOnboarding": true}' ~/.claude.json > "$tmp" || { rm -f "$tmp"; die "~/.claude.json is not valid JSON"; } else jq -n '{"hasCompletedOnboarding": true}' > "$tmp" fi mv "$tmp" ~/.claude.json fi if ! command -v claude >/dev/null 2>&1; then step WARN "token set, but the claude CLI isn't installed"; return 0 fi local method; method="$(claude auth status --json </dev/null 2>/dev/null | jq -r '.authMethod // empty' 2>/dev/null)" || method="" if [[ "$method" == oauth_token ]]; then step OK "oauth_token" else step WARN "token set, but claude reports '${method:-unknown}'" fi } # `code` only works from a VS Code terminal (Remote-SSH, devcontainer, codespace). install_vscode_extension() { CURRENT="VS Code extension" local exts if ! command -v code >/dev/null 2>&1 || ! exts="$(code --list-extensions </dev/null 2>/dev/null)"; then step SKIP "'code' not available" hint "in a VS Code terminal: code --install-extension ${VSCODE_EXTENSION}" return 0 fi if print -r -- "$exts" | grep -qix "$VSCODE_EXTENSION"; then step OK "already installed" elif code --install-extension "$VSCODE_EXTENSION" </dev/null >/dev/null 2>&1; then step OK "installed ${VSCODE_EXTENSION}" else step WARN "install failed; retry: code --install-extension ${VSCODE_EXTENSION}" fi } main() { # Run first -- but not during a dry run (DOTFILES_RESOLVE_ONLY promises no changes). [[ -n "${DOTFILES_RESOLVE_ONLY:-}" ]] || run_cde_init umask 077 # after cde-init, which creates shared workspace files os_guard select_repo_ref [[ -n "${DOTFILES_RESOLVE_ONLY:-}" ]] && { print -r -- "REPO_RAW=${REPO_RAW}"; return 0; } install_omz fetch_configs install_ai_rules install_zsh_plugins install_vim_plugins setup_claude_auth install_vscode_extension print -r -- "" print -r -- "Next: start a fresh shell ('exec zsh'); reload tmux with 'tmux source-file ~/.tmux.conf'." } main
